#include <iostream>
#include <vector>
#include <queue>
#include <chrono>
using namespace std;

class Graph {
    int V;
    vector<vector<int>> adj;

public:
    Graph(int v) : V(v), adj(v) {}

    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    void DFS(int v, vector<bool>& visited) {
        visited[v] = true;
        cout << v << " ";

        for (int n : adj[v])
            if (!visited[n])
                DFS(n, visited);
    }

    void BFS(int start) {
        vector<bool> visited(V, false);
        queue<int> q;

        visited[start] = true;
        q.push(start);

        while (!q.empty()) {
            int v = q.front();
            q.pop();
            cout << v << " ";

            for (int n : adj[v]) {
                if (!visited[n]) {
                    visited[n] = true;
                    q.push(n);
                }
            }
        }
    }
};

int main() {
    int V, E;
    cout << "Vertices: ";
    cin >> V;

    Graph g(V);

    cout << "Edges: ";
    cin >> E;

    cout << "Enter edges:\n";
    for (int i = 0; i < E; i++) {
        int u, v;
        cin >> u >> v;
        g.addEdge(u, v);
    }

    int start;
    cout << "Starting vertex: ";
    cin >> start;

    // DFS
    vector<bool> visited(V, false);
    auto t1 = chrono::high_resolution_clock::now();

    cout << "DFS: ";
    g.DFS(start, visited);

    auto t2 = chrono::high_resolution_clock::now();

    // BFS
    auto t3 = chrono::high_resolution_clock::now();

    cout << "\nBFS: ";
    g.BFS(start);

    auto t4 = chrono::high_resolution_clock::now();

    cout << "\n\nDFS Time: "
         << chrono::duration_cast<chrono::nanoseconds>(t2 - t1).count()
         << " ns";

    cout << "\nBFS Time: "
         << chrono::duration_cast<chrono::nanoseconds>(t4 - t3).count()
         << " ns";

    return 0;
}
