#include <iostream>
#include <chrono>

using namespace std;
using namespace std::chrono;

// Iterative Factorial — O(n) time, O(1) space
unsigned long long factorialIterative(int n) {
    unsigned long long res = 1;
    for (int i = 2; i <= n; ++i) res *= i;
    return res;
}

// Recursive Factorial — O(n) time, O(n) space
unsigned long long factorialRecursive(int n) {
    return (n <= 1) ? 1 : n * factorialRecursive(n - 1);
}

// Generic benchmark helper to avoid repeated timing logic
template <typename Func>
void measure(Func func, int n, const string& label) {
    auto start = high_resolution_clock::now();
    unsigned long long res = func(n);
    auto end = high_resolution_clock::now();

    cout << label << " Result : " << res << "\n"
         << label << " Time   : " << duration<double, nano>(end - start).count() << " ns\n";
}

int main() {
    int n;
    cout << "Enter a non-negative integer (max 20): ";
    if (!(cin >> n) || n < 0) {
        cerr << "Invalid input!\n";
        return 1;
    }

    cout << "\n--- Results for " << n << "! ---\n";
    measure(factorialIterative, n, "Iterative");
    cout << "-------------------------------\n";
    measure(factorialRecursive, n, "Recursive");

    return 0;
}
